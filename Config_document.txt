var sections = {};

function loadInputJson(inputJson) {

    if (!inputJson) {
        return;
    }

    inputJson = String(inputJson)
        .replace(/^\uFEFF/, "")
        .trim();

    if (inputJson === "") {
        return;
    }

    var data;

    try {

        out.println("JSON Length: " + inputJson.length);

        data = JSON.parse(inputJson);

    } catch (e) {

        out.println("JSON.parse failed");
        out.println("Error: " + e);
        out.println(inputJson);

        return;
    }

    if (!Array.isArray(data)) {
        data = [data];
    }

    for (var i = 0; i < data.length; i++) {

        var item = data[i];

        if (!item) {
            continue;
        }

        if (item.glazingobj !== undefined) {
            continue;
        }

        var id = item.id;

        if (!id) {
            continue;
        }

        var value = item.value;
        var desc = item.desc;
        var text = item.text;

        if (
            value === null ||
            value === undefined ||
            value === "null" ||
            value === "undefined" ||
            value === "NONE"
        ) {
            value = "";
        }

        if (desc == null) {
            desc = "";
        }

        if (text == null) {
            text = "";
        }

        value = String(value);
        desc = String(desc);
        text = String(text);

        var finalValue = text || desc || value;

        finalValue = finalValue.replace(/\\/g, "");

        sections[id] = finalValue;
    }
}
