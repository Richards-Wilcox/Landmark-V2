var sections = {};

function loadInputJson(inputJson) {

    if (!inputJson || inputJson.trim() === "") {
        return;
    }

    inputJson = inputJson.trim().replace(/(\r\n|\n|\r|\t)/gm, "");
    inputJson = inputJson.replace(/^\[|\]$/g, "");
    inputJson = inputJson.replace(/&quot;/g, '"');

    var objects = inputJson.split('},{');

    if (objects.length > 0) {
        objects[0] = objects[0].replace('{', '');
        objects[objects.length - 1] =
            objects[objects.length - 1].replace('}', '');
    }

    for (var i = 0; i < objects.length; i++) {

        var obj = objects[i];

        if (obj.indexOf('glazingobj') !== -1) {
            continue;
        }

        var keyValuePairs = obj.split(',');

        var id = '';
        var value = '';
        var desc = '';
        var text = '';

        for (var j = 0; j < keyValuePairs.length; j++) {

            var pair = keyValuePairs[j].split(':');

            var key = pair[0].trim();

            if (key.charAt(0) == '&') {
                key = key.substring(6, key.length - 6);
            }

            var val = pair.slice(1).join(':').trim();

            if (val.charAt(0) == '&') {
                val = val.substring(6, val.length - 6);
            }

            if (key === 'id') {
                id = val;
            }

            if (key === 'value') {
                value = val;

                if (
                    value === 'null' ||
                    value === 'undefined' ||
                    value === 'NONE'
                ) {
                    value = '';
                }
            }

            if (key === 'desc') {
                desc = val;
            }

            if (key === 'text') {
                text = val;
            }
        }

        sections[id] =
            text !== ''
                ? text
                : (desc !== '' ? desc : value);

        if (sections[id]) {
            sections[id] = sections[id].replace(/\\/g, '');
        }
    }
}
