var sections = {};

function loadInputJson(inputJson) {

    if (!inputJson) {
        return;
    }

    inputJson = String(inputJson).trim();

    if (inputJson === "") {
        return;
    }

    inputJson = inputJson.replace(/&quot;/g, '"');

    inputJson = inputJson
        .replace(/"\s*i\s*d\s*"/g, '"id"')
        .replace(/"\s*v\s*a\s*l\s*u\s*e\s*"/g, '"value"')
        .replace(/"\s*d\s*e\s*s\s*c\s*"/g, '"desc"')
        .replace(/"\s*t\s*e\s*x\s*t\s*"/g, '"text"');

    inputJson = inputJson
        .replace(/\bn\s*u\s*l\s*l\b/gi, 'null')
        .replace(/\bf\s*a\s*l\s*s\s*e\b/gi, 'false')
        .replace(/\bt\s*r\s*u\s*e\b/gi, 'true');

    var data;

    try {
        data = JSON.parse(inputJson);
    }
    catch (e) {

        out.println("JSON.parse failed:");
        out.println(inputJson);
        out.println("Error: " + e);

        return;
    }

    if (!Array.isArray(data)) {
        data = [data];
    }

    for (var i = 0; i < data.length; i++) {

        var item = data[i];

        if (!item) {
            continue;
        }

        if (item.glazingobj !== undefined) {
            continue;
        }

        var id = item.id;

        if (!id) {
            continue;
        }

        var value = item.value;
        var desc = item.desc;
        var text = item.text;

        if (
            value === null ||
            value === undefined ||
            value === 'null' ||
            value === 'undefined' ||
            value === 'NONE'
        ) {
            value = '';
        }

        if (
            desc === null ||
            desc === undefined
        ) {
            desc = '';
        }

        if (
            text === null ||
            text === undefined
        ) {
            text = '';
        }

        value = String(value);
        desc = String(desc);
        text = String(text);

        var finalValue = '';

        if (text !== '') {
            finalValue = text;
        }
        else if (desc !== '') {
            finalValue = desc;
        }
        else {
            finalValue = value;
        }

        finalValue = finalValue.replace(/\\/g, '');

        sections[id] = finalValue;
    }
}
